class Autoencoder(nn.Module):
    def __init__(self, bottleneck=8):
        super().__init__()
        self.encoder = nn.Sequential(nn.Linear(29, 16), nn.ReLU(),
                                     nn.Linear(16, bottleneck), nn.ReLU())
        self.decoder = nn.Sequential(nn.Linear(bottleneck, 16), nn.ReLU(),
                                     nn.Linear(16, 29))
    def forward(self, x):
        return self.decoder(self.encoder(x))

def reconstruction_errors(model, values):
    model.eval()
    scores = []
    with torch.no_grad():
        for batch in torch.from_numpy(values).split(4096):
            scores.append(((model(batch) - batch)**2).mean(dim=1).numpy())
    return np.concatenate(scores)

2. The hybrid, step by step¶

GMM scores and AE errors have different units, so adding them directly would give arbitrary influence.

  1. Compress the GMM score with asinh (works for negative and positive values).
  2. Compress nonnegative AE error with log1p.
  3. For each transformed score, subtract its normal-validation median and divide by its interquartile range (75th percentile minus 25th percentile).
  4. hybrid = 0.5 * normalized_GMM + 0.5 * normalized_AE.

Median and IQR are less sensitive to extreme values than mean and standard deviation. The log-like transforms reduce extreme tail domination. Monotone transforms preserve each individual ranking, but their combination can change rankings. The hybrid remains a score, not a fraud probability. Weights stay 0.5/0.5. We do not optimize weights using fraud labels.

normal = y_validation == 0
normalization = {}
def normalize(values, name, fit=False):
    t = np.arcsinh(values) if name == "GMM" else np.log1p(values)
    if fit:
        ref = t[normal]
        q25, median, q75 = np.percentile(ref, [25, 50, 75])
        normalization[name] = {"median": float(median), "iqr": float(max(q75-q25, 1e-8))}
    settings = normalization[name]
    return (t-settings["median"])/settings["iqr"]
validation_scores["Hybrid"] = 0.5*normalize(validation_scores["GMM"], "GMM", True) +                               0.5*normalize(validation_scores["denoising_ae"], "AE", True)
budgets = [0.001, 0.005, 0.01]
thresholds, val_rows = {}, []
for name, scores in validation_scores.items():
    assert np.isfinite(scores).all()
    thresholds[name] = {}
    for budget in budgets:
        cutoff = float(np.quantile(scores[normal], 1-budget, method="higher"))
        observed = float(np.mean(scores[normal] > cutoff))
        assert observed <= budget
        thresholds[name][str(budget)] = cutoff
        val_rows.append({"model": name, "budget": budget, "cutoff": cutoff, "validation_fpr": observed})

3. Evaluate the later mixed test period¶

Precision = fraud found / all alerts. Recall = fraud found / all actual fraud. FPR = legitimate transactions alerted / all legitimate transactions.

PR AUC here means trapezoidal area for continuity with Phase 1. We also report AP, a step-weighted summary of the PR curve. AP and trapezoidal PR AUC are not identical. Precision@100 means fraud among the top 100 scores divided by 100. It is a capacity policy for this whole test batch, not 100 reviews per day.

ranking_rows, policy_rows, capacity_rows = [], [], []
for name, scores in test_scores.items():
    assert len(scores) == len(y_test) and np.isfinite(scores).all()
    precision, recall, _ = precision_recall_curve(y_test, scores)
    order = np.argsort(-scores, kind="stable")
    ranking_rows.append({"model": name, "pr_auc_trapezoidal": auc(recall, precision),
                         "average_precision": average_precision_score(y_test, scores),
                         "precision_at_100": float(y_test[order[:100]].mean()),
                         "frauds_at_100": int(y_test[order[:100]].sum())})
    for budget in budgets:
        cutoff = thresholds[name][str(budget)]
        tn, fp, fn, tp = confusion_matrix(y_test, scores > cutoff, labels=[0,1]).ravel()
        assert tn+fp+fn+tp == len(y_test) and tp+fn == y_test.sum()
        policy_rows.append({"model": name, "validation_budget": budget, "cutoff": cutoff,
                            "tn": int(tn), "fp": int(fp), "fn": int(fn), "tp": int(tp),
                            "test_recall": tp/(tp+fn), "test_precision": tp/max(tp+fp,1),
                            "test_fpr": fp/(fp+tn), "alerts": int(tp+fp)})
    for k in [50,100,200,500]:
        found = int(y_test[order[:k]].sum())
        capacity_rows.append({"model": name, "reviews": k, "frauds_found": found,
                              "precision": found/k, "recall": found/int(y_test.sum())})

8. Save the conclusion and checks¶

The same validation FPR target can yield different test FPRs. A future rate is not guaranteed by a past cutoff. With only 74 test frauds, treat small differences cautiously. Two seeds cannot establish statistical superiority. Bootstrap below resamples the two labels separately, preserving their observed counts. It estimates conditional sampling uncertainty for paired AP differences; it ignores model refitting and temporal dependence.
